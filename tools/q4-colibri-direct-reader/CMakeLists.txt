set(TARGET llama-q4-colibri-direct-reader)
find_library(Q4_COLIBRI_LIBURING uring)
if (NOT Q4_COLIBRI_LIBURING)
    message(STATUS "liburing not found: ${TARGET} is not built")
    return()
endif()
find_package(CUDAToolkit REQUIRED)
add_executable(${TARGET} q4-colibri-direct-reader.cpp ../../examples/gguf-hash/deps/sha256/sha256.c)
target_include_directories(${TARGET} PRIVATE ../../vendor ../../examples/gguf-hash/deps ../../examples/gguf-hash/deps/sha256)
target_link_libraries(${TARGET} PRIVATE ${Q4_COLIBRI_LIBURING} CUDA::cudart)
target_compile_features(${TARGET} PRIVATE cxx_std_17)
install(TARGETS ${TARGET} RUNTIME)
