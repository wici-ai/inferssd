if(GGML_HIP)
    # The probe in HIP, with the runtime's names from the backend's vendor header (gpu-runtime.h), for the
    # backend's GPU architectures.
    get_target_property(CMAKE_HIP_ARCHITECTURES ggml-hip HIP_ARCHITECTURES)
    enable_language(HIP)
    find_package(hip REQUIRED CONFIG HINTS "${CMAKE_HIP_COMPILER_ROCM_ROOT}")
    set_source_files_properties(../memprobe.cu PROPERTIES LANGUAGE HIP)
    add_executable(inferssd-memprobe ../memprobe.cu)
    target_compile_definitions(inferssd-memprobe PRIVATE GGML_USE_HIP)
    target_link_libraries(inferssd-memprobe PRIVATE hip::host)
else()
    enable_language(CUDA)
    add_executable(inferssd-memprobe ../memprobe.cu)
    target_link_libraries(inferssd-memprobe PRIVATE CUDA::cudart)
endif()
target_include_directories(inferssd-memprobe PRIVATE ${PROJECT_SOURCE_DIR}/ggml/src)
install(TARGETS inferssd-memprobe RUNTIME)
