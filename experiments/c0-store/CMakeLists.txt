cmake_minimum_required(VERSION 3.19)
project(c0_store_prototype LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
add_executable(test-c0-store test.cpp)
target_compile_features(test-c0-store PRIVATE cxx_std_17)
target_include_directories(test-c0-store PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(test-c0-store PRIVATE -Wall -Wextra -Werror -Wno-unused-function)
option(C0_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(C0_SANITIZERS)
  target_compile_options(test-c0-store PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(test-c0-store PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
find_package(Threads REQUIRED)
foreach(variant IN ITEMS cpu required)
  add_library(c0-shared-${variant} SHARED ../../ggml/src/ggml-ssd/store/store.cpp ../../ggml/src/ggml-ssd/store/parser.cpp)
  target_compile_features(c0-shared-${variant} PRIVATE cxx_std_17)
  target_include_directories(c0-shared-${variant} PUBLIC
      ../../ggml/src/ggml-ssd/store)
  target_link_libraries(c0-shared-${variant} PRIVATE Threads::Threads)
  add_library(c0-mock-${variant} MODULE mock-provider.cpp)
  target_link_libraries(c0-mock-${variant} PRIVATE c0-shared-${variant})
  target_compile_features(c0-mock-${variant} PRIVATE cxx_std_17)
  add_executable(test-c0-shared-${variant} test-shared.cpp)
  target_link_libraries(test-c0-shared-${variant} PRIVATE c0-shared-${variant} Threads::Threads ${CMAKE_DL_LIBS})
  target_compile_features(test-c0-shared-${variant} PRIVATE cxx_std_17)
  if(variant STREQUAL "required")
    target_compile_definitions(c0-shared-${variant} PRIVATE GGML_C0_REQUIRE_CUDA_PROVIDER)
    target_compile_definitions(test-c0-shared-${variant} PRIVATE C0_EXPECT_PROVIDER)
  endif()
  foreach(target IN ITEMS c0-shared-${variant} c0-mock-${variant} test-c0-shared-${variant})
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-unused-function)
    if(C0_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endforeach()
  foreach(mode IN ITEMS normal concurrent throw disabled late unload page-cache shape-k3 half-success half-failure)
    add_test(NAME c0-shared-${variant}-${mode}
        COMMAND test-c0-shared-${variant} ${mode} $<TARGET_FILE:c0-mock-${variant}>)
    set_tests_properties(c0-shared-${variant}-${mode} PROPERTIES TIMEOUT 30)
  endforeach()
endforeach()
target_link_libraries(test-c0-store PRIVATE c0-shared-cpu)
