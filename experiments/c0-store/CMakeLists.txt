cmake_minimum_required(VERSION 3.19)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(c0_store_tests LANGUAGES CXX)
endif()
# Immutable test oracle only. The production store never includes these files.
file(GLOB C0_ORACLE_INCLUDES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/oracle/*.inc")
foreach(reference IN LISTS C0_ORACLE_INCLUDES)
  get_filename_component(name "${reference}" NAME)
  configure_file("${reference}" "${CMAKE_CURRENT_BINARY_DIR}/generated/${name}" COPYONLY)
endforeach()
add_executable(test-c0-store test.cpp)
target_compile_features(test-c0-store PRIVATE cxx_std_17)
target_include_directories(test-c0-store PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(test-c0-store PRIVATE -Wall -Wextra -Werror -Wno-unused-function)
option(C0_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(C0_SANITIZERS)
  target_compile_options(test-c0-store PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(test-c0-store PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
find_package(Threads REQUIRED)
foreach(variant IN ITEMS cpu required)
  add_library(c0-shared-${variant} SHARED ../../ggml/src/ggml-ssd/store/store.cpp ../../ggml/src/ggml-ssd/store/parser.cpp)
  target_compile_features(c0-shared-${variant} PRIVATE cxx_std_17)
  target_include_directories(c0-shared-${variant} PUBLIC
      ../../ggml/src/ggml-ssd/store)
  target_link_libraries(c0-shared-${variant} PRIVATE Threads::Threads)
  add_library(c0-mock-${variant} MODULE mock-provider.cpp)
  target_link_libraries(c0-mock-${variant} PRIVATE c0-shared-${variant})
  target_compile_features(c0-mock-${variant} PRIVATE cxx_std_17)
  add_executable(test-c0-shared-${variant} test-shared.cpp)
  add_dependencies(test-c0-shared-${variant} c0-mock-${variant})
  target_link_libraries(test-c0-shared-${variant} PRIVATE c0-shared-${variant} Threads::Threads ${CMAKE_DL_LIBS})
  target_compile_features(test-c0-shared-${variant} PRIVATE cxx_std_17)
  if(variant STREQUAL "required")
    target_compile_definitions(c0-shared-${variant} PRIVATE GGML_SSD_REQUIRE_CUDA_PROVIDER)
    target_compile_definitions(test-c0-shared-${variant} PRIVATE C0_EXPECT_PROVIDER)
  endif()
  foreach(target IN ITEMS c0-shared-${variant} c0-mock-${variant} test-c0-shared-${variant})
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-unused-function)
    if(C0_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endforeach()
  foreach(mode IN ITEMS normal concurrent throw disabled late unload page-cache shape-k3 half-success half-failure)
    add_test(NAME c0-shared-${variant}-${mode}
        COMMAND test-c0-shared-${variant} ${mode} $<TARGET_FILE:c0-mock-${variant}>)
    set_tests_properties(c0-shared-${variant}-${mode} PROPERTIES TIMEOUT 30)
  endforeach()
endforeach()
if(TARGET ggml-ssd-c0-store)
  set(C0_PARSER_TEST_LIBRARY ggml-ssd-c0-store)
else()
  set(C0_PARSER_TEST_LIBRARY c0-shared-cpu)
endif()
target_link_libraries(test-c0-store PRIVATE ${C0_PARSER_TEST_LIBRARY})
add_test(NAME c0-store COMMAND test-c0-store)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(test-c0-direct test-direct.cpp)
  target_compile_features(test-c0-direct PRIVATE cxx_std_17)
  target_include_directories(test-c0-direct PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(test-c0-direct PRIVATE c0-shared-cpu)

    target_compile_options(test-c0-direct PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(test-c0-direct PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME c0-direct COMMAND test-c0-direct)
  add_library(c0-raw-direct-shim SHARED raw-direct-shim.cpp)
  target_compile_features(c0-raw-direct-shim PRIVATE cxx_std_17)
  target_link_libraries(c0-raw-direct-shim PRIVATE c0-shared-cpu)
  target_include_directories(c0-raw-direct-shim PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
endif()
add_executable(test-c0-config test-config.cpp)
target_compile_features(test-c0-config PRIVATE cxx_std_17)
target_include_directories(test-c0-config PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(test-c0-config PRIVATE -Wall -Wextra -Werror -Wno-unused-function)

target_link_libraries(test-c0-config PRIVATE ${C0_PARSER_TEST_LIBRARY})

target_link_libraries(test-c0-parser-hooks PRIVATE c0-shared-cpu)

  target_link_options(test-c0-parser-hooks PRIVATE -fsanitize=address,undefined)

  target_link_options(test-c0-parser-failures PRIVATE -fsanitize=address,undefined)

  target_compile_features(test-c0-linked PRIVATE cxx_std_17)
