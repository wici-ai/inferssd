cmake_minimum_required(VERSION 3.19)
project(c0_store_prototype LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract.py" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
add_executable(test-c0-store test.cpp)
target_compile_features(test-c0-store PRIVATE cxx_std_17)
target_include_directories(test-c0-store PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(test-c0-store PRIVATE -Wall -Wextra -Werror -Wno-unused-function)
option(C0_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(C0_SANITIZERS)
  target_compile_options(test-c0-store PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(test-c0-store PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME c0-store COMMAND test-c0-store)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(test-c0-direct test-direct.cpp)
  target_compile_features(test-c0-direct PRIVATE cxx_std_17)
  target_include_directories(test-c0-direct PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  add_test(NAME c0-direct COMMAND test-c0-direct)
  add_library(c0-raw-direct-shim SHARED raw-direct-shim.cpp)
  target_compile_features(c0-raw-direct-shim PRIVATE cxx_std_17)
  target_include_directories(c0-raw-direct-shim PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
endif()
add_executable(test-c0-config test-config.cpp)
target_compile_features(test-c0-config PRIVATE cxx_std_17)
target_include_directories(test-c0-config PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(test-c0-config PRIVATE -Wall -Wextra -Werror -Wno-unused-function)
if(C0_SANITIZERS)
  target_compile_options(test-c0-config PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(test-c0-config PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME c0-config COMMAND test-c0-config)
