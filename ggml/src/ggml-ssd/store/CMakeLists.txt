# One owner DSO; copying a static archive into both consumers is unsupported.
if (NOT GGML_CPU)
    message(FATAL_ERROR "LLAMA_SSD_STORE requires GGML_CPU=ON for the CPU sink executor")
endif()
if (GGML_BACKEND_DL OR GGML_CPU_ALL_VARIANTS)
    message(FATAL_ERROR "LLAMA_SSD_STORE requires GGML_BACKEND_DL=OFF and GGML_CPU_ALL_VARIANTS=OFF; CPU sink entry points are not exported through the dynamic registry yet")
endif()
include(GNUInstallDirs)
find_package(Threads REQUIRED)
add_library(ggml-ssd-c0-store SHARED store.cpp parser.cpp index.cpp parser.h index.h source.h store.h catalog.h legacy-index.h legacy-config.h)
target_compile_features(ggml-ssd-c0-store PRIVATE cxx_std_17)
target_link_libraries(ggml-ssd-c0-store PRIVATE Threads::Threads)
set_target_properties(ggml-ssd-c0-store PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
target_include_directories(ggml-ssd-c0-store PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if (GGML_CUDA)
    target_compile_definitions(ggml-ssd-c0-store PRIVATE GGML_SSD_REQUIRE_CUDA_PROVIDER)
endif()
install(TARGETS ggml-ssd-c0-store LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
